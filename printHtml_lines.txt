     8	export function requireStableStringPart(value: unknown, context: string): string {
     9	  if (typeof value !== 'string') {
    10	    throw new Error(`معرّف غير صالح [${context}]: القيمة غائبة أو ليست نصاً (النوع: ${typeof value})`);
    11	  }
    12	  const trimmed = value.trim();
    13	  if (!trimmed) {
    14	    throw new Error(`معرّف غير صالح [${context}]: القيمة نص فارغ`);
    15	  }
    16	  return trimmed;
    17	}

   479	  // Preflight validation
   480	  const sourceRecordIds: string[][] = [];
   481	  if (!Array.isArray(data.tables)) throw new Error('data.tables is missing or invalid');
   482	  data.tables.forEach((t, tIdx) => {
   483	    if (!Array.isArray(t.recordIds)) throw new Error(`الجدول ${tIdx} يفتقد recordIds`);
   484	    if (t.recordIds.length !== t.rows.length) throw new Error(`الجدول ${tIdx}: عدد recordIds (${t.recordIds.length}) لا يطابق rows (${t.rows.length})`);
   485	    const tableIds: string[] = [];

   488	      if (typeof id !== 'string') throw new Error(`الجدول ${tIdx} الصف ${rIdx}: الهوية ليست نصاً`);
